% ------------------------------------------------------------
% RISC-V encoding environment
%
% Usage:
%
%   \begin{riscvencoding}[R-type]{mulh}{32}
%     \field{0000001}{7}
%     \field{rs2}{5}
%     \field{rs1}{5}
%     \field{001}{3}
%     \field{rd}{5}
%     \field{0110011}{7}
%   \end{riscvencoding}
%
%   \begin{riscvencoding}[C-type]{C.ADDI}{16}
%     \field{01}{2}
%     \field{000}{3}
%     \field{imm[5]}{1}
%     \field{rs1}{5}
%     \field{imm[4:0]}{5}
%   \end{riscvencoding}
%
% Arguments:
%
%   #1 = optional instruction format
%   #2 = instruction name
%   #3 = instruction width in bits
%
% The complete diagram occupies \linewidth.
% ------------------------------------------------------------

\newenvironment{riscvencoding}[3][]{%

  % Save instruction width.
  \riscvinstructionbits=#3\relax

  % Reserve a small fixed column for the instruction format/name.
  % The encoding itself uses the remaining width, so the diagram
  % does not overrun the instruction content column.
  \setlength{\riscvdiagramwidth}{%
    \dimexpr\linewidth-4.8em\relax
  }%

  % One bit occupies an equal fraction of the encoding width.
  \setlength{\riscvbitwidth}{%
    \dimexpr\riscvdiagramwidth/#3\relax
  }%

  % Position of the format/name annotation, in bit coordinates.
  \pgfmathsetmacro{\riscvformatpos}{\the\riscvinstructionbits+0.7}%

  % Start at the most-significant bit.
  \riscvencodingpos=0%

  % No fields emitted yet.
  \riscvencodingbits=0%

  % Save format and instruction name.
  \def\riscvformat{#1}%
  \def\riscvinstruction{#2}%

  \begin{center}
  \begin{tikzpicture}[riscv/encoding]

}{%

  % Verify that the fields occupy the complete instruction.
  \ifnum\riscvencodingbits=\riscvinstructionbits
  \else
    \PackageError{riscvencoding}
      {Instruction encoding contains \the\riscvencodingbits\space bits}
      {The encoding must contain exactly
       \the\riscvinstructionbits\space bits.}%
  \fi

  % Optional instruction format and name.
  \ifx\riscvformat\empty
  \else
    \node[
      riscv/format
    ] at (
      \riscvformatpos,
      0.52
    ) {\riscvformat};

    \node[
      riscv/format
    ] at (
      \riscvformatpos,
      0.16
    ) {\instr{\riscvinstruction}};
  \fi

  \end{tikzpicture}
  \end{center}
}


% ------------------------------------------------------------
% \field{label}{width}
%
% #1 = field label / encoded value
% #2 = field width in bits
%
% Examples:
%
%   \field{0000001}{7}
%   \field{rs2}{5}
%   \field{imm[4:0]}{5}
%
% The label is rendered exactly as supplied.
% ------------------------------------------------------------

\newcommand{\field}[2]{%

  % ----------------------------------------------------------
  % Calculate field boundaries
  % ----------------------------------------------------------

  % Position immediately after this field.
  \pgfmathtruncatemacro{\riscvendpos}{%
    \riscvencodingpos + #2%
  }%

  % Most-significant bit of this field.
  \pgfmathtruncatemacro{\riscvstartbit}{%
    \riscvinstructionbits - 1 - \riscvencodingpos
  }%

  % Least-significant bit of this field.
  \pgfmathtruncatemacro{\riscvendbit}{%
    \riscvinstructionbits - \riscvendpos
  }%

  % ----------------------------------------------------------
  % Validate field
  % ----------------------------------------------------------

  \ifnum\riscvendpos>\riscvinstructionbits
    \PackageError{riscvencoding}
      {Encoding field exceeds instruction width}
      {A field extends beyond the
       \the\riscvinstructionbits-bit instruction.}%
  \fi

  % ----------------------------------------------------------
  % Field rectangle
  % ----------------------------------------------------------

  \draw[
    riscv/field
  ]
    (\riscvencodingpos,0)
    rectangle
    (\riscvendpos,0.72);

  % ----------------------------------------------------------
  % Field label
  %
  % The field is 0.72 units high, so 0.36 is its vertical
  % center.
  % ----------------------------------------------------------

  \node[
    riscv/value
  ] at (
    \riscvencodingpos + #2/2,
    0.36
  ) {#1};

  % ----------------------------------------------------------
  % Field bit numbers
  %
  % Both numbers are placed inside the field, slightly inset
  % from its boundaries.
  %
  % Example:
  %
  %   31        25 24        20
  %   ┌───────────┬────────────┐
  %
  % The left number uses south-west anchoring and the right
  % number uses south-east anchoring, so adjacent numbers
  % remain separated.
  % ----------------------------------------------------------

  % Most-significant bit.
  \node[
    riscv/bits,
    anchor=south west,
    xshift=2pt
  ] at (
    \riscvencodingpos,
    0.98
  ) {\riscvstartbit};

  % Least-significant bit.
  \node[
    riscv/bits,
    anchor=south east,
    xshift=-2pt
  ] at (
    \riscvendpos,
    0.98
  ) {\riscvendbit};

  % ----------------------------------------------------------
  % Advance to next field
  % ----------------------------------------------------------

  \riscvencodingpos=\riscvendpos

  \advance\riscvencodingbits by #2\relax
}

\section{Introduction}

The RISC-V instruction set architecture defines a family of
open standard instruction set architectures for general-purpose
computing.

This document describes the RISC-V base instruction sets and
extensions. The instruction descriptions are organized by
architectural extension and use a common notation for operands,
operations, and instruction encodings.

% ------------------------------------------------------------
% Base ISA
% ------------------------------------------------------------

\section{Base ISA}

\subsection{I --- RV32I / RV64I / RV128I --- Base Integer ISA}

The I extension defines the base integer instruction set.
It is available in RV32I, RV64I, and RV128I variants.

\subsection{E --- RV32E / RV64E --- Reduced Integer ISA, 16 Registers}

The E extension defines a reduced integer instruction set with
16 integer registers.

% ------------------------------------------------------------
% Extensions
% ------------------------------------------------------------

\section{Extensions}

\subsection{M --- Integer Multiply/Divide}

This chapter describes the standard integer multiplication and
division instruction extension, which is named {\sf M} and contains
instructions that multiply or divide values held in two integer
registers.

\subsubsection{\instr{mul} --- Multiply Integer $\times$ Integer}

\begin{instructionrow}{Synopsis}

\begin{lstlisting}
mul rd, rs1, rs2
\end{lstlisting}
\end{instructionrow}

\begin{instructionrow}{Description}
The \instr{mul} instruction multiplies the values in the source
registers \instr{rs1} and \instr{rs2} and stores the lower
\texttt{XLEN} bit result to the description registers \instr{rd}.
\end{instructionrow}

\begin{instructionrow}{Operation}
\[
\rd = \rsone\,\mul\,\rstwo
\]
\end{instructionrow}

\begin{instructionrow}{Encoding}

\begin{center}
  \begin{riscvencoding}[R-type]{mul}{32}
    \field{0000001}{7}
    \field{rs2}{5}
    \field{rs1}{5}
    \field{000}{3}
    \field{rd}{5}
    \field{0110011}{7}
  \end{riscvencoding}
  \captionof{figure}{mul instruction encoding.}
  \label{fig:mulh-encoding}
\end{center}
\end{instructionrow}


% ------------------------------------------------------------
% Instruction: mulh
% ------------------------------------------------------------

\subsubsection{\instr{mulh} --- Multiply High Signed $\times$ Signed}

\begin{instructionrow}{Synopsis}

\begin{lstlisting}
mulh rd, rs1, rs2
\end{lstlisting}
\end{instructionrow}

\begin{instructionrow}{Description}

The \instr{mulh} instruction multiplies the values in the
source registers \instr{rs1} and \instr{rs2}, treating both
operands as signed two's-complement integers. The upper
\mbox{\texttt{XLEN}} bits of the resulting
$2\times\mbox{\texttt{XLEN}}$-bit product are written to the
destination register \instr{rd}.

The lower \mbox{\texttt{XLEN}} bits of the product are discarded.
The operation therefore provides the high half of a signed
integer multiplication without requiring the full product to
be represented in a single integer register.

\end{instructionrow}

\begin{instructionrow}{Operation}

\[
\rd =
\left(
  \rsone\,\smul\,\rstwo
\right)
\signedshift XLEN
\]

Equivalently, if the signed values represented by
\instr{rs1} and \instr{rs2} are denoted by $a$ and $b$, then
\instr{rd} receives the upper \mbox{\texttt{XLEN}} bits of the
$2\times\mbox{\texttt{XLEN}}$-bit signed product $a\times b$.
\end{instructionrow}

\begin{instructionrow}{Encoding}

\begin{center}
  \begin{riscvencoding}[R-type]{mulh}{32}
    \field{0000001}{7}
    \field{rs2}{5}
    \field{rs1}{5}
    \field{001}{3}
    \field{rd}{5}
    \field{0110011}{7}
  \end{riscvencoding}
  \captionof{figure}{mulh instruction encoding.}
  \label{fig:mulh-encoding}
\end{center}
\end{instructionrow}

\begin{instructionrow}{Operands}

\operand{rd}{Destination integer register.}
\operand{rs1}{First source integer register.}
\operand{rs2}{Second source integer register.}
\end{instructionrow}

\begin{instructionrow}{Availability}

RV32M and RV64M.
\end{instructionrow}
